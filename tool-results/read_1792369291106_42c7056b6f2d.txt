    98→/** يفحص الإنجازات المحققة (الصلوات + التحدي المدمج) ويحفظ الجديد منها */
    99→function checkAchievements(logs: Logs, planner: PlannerState | null, best: number): string[] {
   100→  const total = Object.values(logs).reduce((a, log) => a + dayCount(log), 0);
   101→  const fullDays = Object.values(logs).filter((l) => dayCount(l) >= 5).length;
   102→  const streak = currentStreak(logs);
   103→  const bestVal = Math.max(streak, best);
   104→  const fajrCount = Object.values(logs).filter((l) => l.fajr).length;
   105→  const xp = Object.values(logs).reduce(
   106→    (a, log) => a + dayCount(log) * 10 + (dayCount(log) >= 5 ? 25 : 0),
   107→    0
   108→  );
   109→  const unlocked: string[] = [];
   110→  const add = (id: string, cond: boolean) => cond && unlocked.push(id);
   111→  add("first-prayer", total >= 1);
   112→  add("full-day", fullDays >= 1);
   113→  add("streak-3", bestVal >= 3);
   114→  add("streak-7", bestVal >= 7);
   115→  add("streak-30", bestVal >= 30);
   116→  add("prayers-50", total >= 50);
   117→  add("prayers-100", total >= 100);
   118→  add("fajr-master", fajrCount >= 5);
   119→  add("week-perfect", fullDays >= 7);
   120→  add("level-5", xp >= 400);
   121→  // أوسمة التاسكات والتايمر (الدمج مع Challenge Planner)
   122→  if (planner) {
   123→    const m = plannerMetrics(planner);
   124→    add("first-task", m.done >= 1);
   125→    add("task-day", m.fullDays >= 1);
   126→    add("study-1h", m.study >= 3600);
   127→    add("tasks-streak-3", m.best >= 3);
   128→    add("units-10", m.units >= 10);
   129→    add("study-10h", m.study >= 36000);
   130→    add("task-half", m.pct >= 50);
   131→    add("units-50", m.units >= 50);
   132→    add("study-25h", m.study >= 90000);
   133→    add("course-done", m.courseDone);
   134→    add("all-tasks", m.total > 0 && m.done === m.total);
   135→  }
   136→  return unlocked;
   137→}
