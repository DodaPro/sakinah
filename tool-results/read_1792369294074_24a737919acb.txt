    98→    98→/** يفحص الإنجازات المحققة (الصلوات + التحدي المدمج) ويحفظ الجديد منها */
    99→    99→function checkAchievements(logs: Logs, planner: PlannerState | null, best: number): string[] {
   100→   100→  const total = Object.values(logs).reduce((a, log) => a + dayCount(log), 0);
   101→   101→  const fullDays = Object.values(logs).filter((l) => dayCount(l) >= 5).length;
   102→   102→  const streak = currentStreak(logs);
   103→   103→  const bestVal = Math.max(streak, best);
   104→   104→  const fajrCount = Object.values(logs).filter((l) => l.fajr).length;
   105→   105→  const xp = Object.values(logs).reduce(
   106→   106→    (a, log) => a + dayCount(log) * 10 + (dayCount(log) >= 5 ? 25 : 0),
   107→   107→    0
   108→   108→  );
   109→   109→  const unlocked: string[] = [];
   110→   110→  const add = (id: string, cond: boolean) => cond && unlocked.push(id);
   111→   111→  add("first-prayer", total >= 1);
   112→   112→  add("full-day", fullDays >= 1);
   113→   113→  add("streak-3", bestVal >= 3);
   114→   114→  add("streak-7", bestVal >= 7);
   115→   115→  add("streak-30", bestVal >= 30);
   116→   116→  add("prayers-50", total >= 50);
   117→   117→  add("prayers-100", total >= 100);
   118→   118→  add("fajr-master", fajrCount >= 5);
   119→   119→  add("week-perfect", fullDays >= 7);
   120→   120→  add("level-5", xp >= 400);
   121→   121→  // أوسمة التاسكات والتايمر (الدمج مع Challenge Planner)
   122→   122→  if (planner) {
   123→   123→    const m = plannerMetrics(planner);
   124→   124→    add("first-task", m.done >= 1);
   125→   125→    add("task-day", m.fullDays >= 1);
   126→   126→    add("study-1h", m.study >= 3600);
   127→   127→    add("tasks-streak-3", m.best >= 3);
   128→   128→    add("units-10", m.units >= 10);
   129→   129→    add("study-10h", m.study >= 36000);
   130→   130→    add("task-half", m.pct >= 50);
   131→   131→    add("units-50", m.units >= 50);
   132→   132→    add("study-25h", m.study >= 90000);
   133→   133→    add("course-done", m.courseDone);
   134→   134→    add("all-tasks", m.total > 0 && m.done === m.total);
   135→   135→  }
   136→   136→  return unlocked;
   137→   137→}
