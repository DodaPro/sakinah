    79→/** يفحص الإنجازات المحققة ويحفظ الجديد منها */
    80→function checkAchievements(logs: Logs): string[] {
    81→  const total = Object.values(logs).reduce((a, log) => a + dayCount(log), 0);
    82→  const fullDays = Object.values(logs).filter((l) => dayCount(l) >= 5).length;
    83→  const streak = currentStreak(logs);
    84→  const best = Math.max(streak, loadBestStreak());
    85→  const fajrCount = Object.values(logs).filter((l) => l.fajr).length;
    86→  const xp = Object.values(logs).reduce(
    87→    (a, log) => a + dayCount(log) * 10 + (dayCount(log) >= 5 ? 25 : 0),
    88→    0
    89→  );
    90→  const unlocked: string[] = [];
    91→  const add = (id: string, cond: boolean) => cond && unlocked.push(id);
    92→  add("first-prayer", total >= 1);
    93→  add("full-day", fullDays >= 1);
    94→  add("streak-3", best >= 3);
    95→  add("streak-7", best >= 7);
    96→  add("streak-30", best >= 30);
    97→  add("prayers-50", total >= 50);
    98→  add("prayers-100", total >= 100);
    99→  add("fajr-master", fajrCount >= 5);
   100→  add("week-perfect", fullDays >= 7);
   101→  add("level-5", xp >= 400);
   102→  return unlocked;
   103→}
