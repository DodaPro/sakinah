    80→    80→function checkAchievements(logs: Logs): string[] {
    81→    81→  const total = Object.values(logs).reduce((a, log) => a + dayCount(log), 0);
    82→    82→  const fullDays = Object.values(logs).filter((l) => dayCount(l) >= 5).length;
    83→    83→  const streak = currentStreak(logs);
    84→    84→  const best = Math.max(streak, loadBestStreak());
    85→    85→  const fajrCount = Object.values(logs).filter((l) => l.fajr).length;
    86→    86→  const xp = Object.values(logs).reduce(
    87→    87→    (a, log) => a + dayCount(log) * 10 + (dayCount(log) >= 5 ? 25 : 0),
    88→    88→    0
    89→    89→  );
    90→    90→  const unlocked: string[] = [];
    91→    91→  const add = (id: string, cond: boolean) => cond && unlocked.push(id);
    92→    92→  add("first-prayer", total >= 1);
    93→    93→  add("full-day", fullDays >= 1);
    94→    94→  add("streak-3", best >= 3);
    95→    95→  add("streak-7", best >= 7);
    96→    96→  add("streak-30", best >= 30);
    97→    97→  add("prayers-50", total >= 50);
    98→    98→  add("prayers-100", total >= 100);
    99→    99→  add("fajr-master", fajrCount >= 5);
   100→   100→  add("week-perfect", fullDays >= 7);
   101→   101→  add("level-5", xp >= 400);
   102→   102→  return unlocked;
   103→   103→}
