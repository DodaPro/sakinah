    93→    93→/** يفحص الإنجازات المحققة (الصلوات + التحدي المدمج) ويحفظ الجديد منها */
    94→    94→function checkAchievements(logs: Logs, planner: PlannerState | null, best: number): string[] {
    95→    95→  const total = Object.values(logs).reduce((a, log) => a + dayCount(log), 0);
    96→    96→  const fullDays = Object.values(logs).filter((l) => dayCount(l) >= 5).length;
    97→    97→  const streak = currentStreak(logs);
    98→    98→  const bestVal = Math.max(streak, best);
    99→    99→  const fajrCount = Object.values(logs).filter((l) => l.fajr).length;
   100→   100→  const xp = Object.values(logs).reduce(
   101→   101→    (a, log) => a + dayCount(log) * 10 + (dayCount(log) >= 5 ? 25 : 0),
   102→   102→    0
   103→   103→  );
   104→   104→  const unlocked: string[] = [];
   105→   105→  const add = (id: string, cond: boolean) => cond && unlocked.push(id);
   106→   106→  add("first-prayer", total >= 1);
   107→   107→  add("full-day", fullDays >= 1);
   108→   108→  add("streak-3", bestVal >= 3);
   109→   109→  add("streak-7", bestVal >= 7);
   110→   110→  add("streak-30", bestVal >= 30);
   111→   111→  add("prayers-50", total >= 50);
   112→   112→  add("prayers-100", total >= 100);
   113→   113→  add("fajr-master", fajrCount >= 5);
   114→   114→  add("week-perfect", fullDays >= 7);
   115→   115→  add("level-5", xp >= 400);
   116→   116→  // أوسمة التاسكات والتايمر (الدمج مع Challenge Planner)
   117→   117→  if (planner) {
   118→   118→    const m = plannerMetrics(planner);
   119→   119→    add("first-task", m.done >= 1);
   120→   120→    add("task-day", m.fullDays >= 1);
   121→   121→    add("study-1h", m.study >= 3600);
   122→   122→    add("tasks-streak-3", m.best >= 3);
   123→   123→    add("units-10", m.units >= 10);
   124→   124→    add("study-10h", m.study >= 36000);
   125→   125→    add("task-half", m.pct >= 50);
   126→   126→    add("units-50", m.units >= 50);
   127→   127→    add("study-25h", m.study >= 90000);
   128→   128→    add("course-done", m.courseDone);
   129→   129→    add("all-tasks", m.total > 0 && m.done === m.total);
   130→   130→  }
   131→   131→  return unlocked;
   132→   132→}
