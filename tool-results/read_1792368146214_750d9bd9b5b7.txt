    96→/** يفحص الإنجازات المحققة (الصلوات + التحدي المدمج) ويحفظ الجديد منها */
    97→function checkAchievements(logs: Logs, planner: PlannerState | null, best: number): string[] {
    98→  const total = Object.values(logs).reduce((a, log) => a + dayCount(log), 0);
    99→  const fullDays = Object.values(logs).filter((l) => dayCount(l) >= 5).length;
   100→  const streak = currentStreak(logs);
   101→  const bestVal = Math.max(streak, best);
   102→  const fajrCount = Object.values(logs).filter((l) => l.fajr).length;
   103→  const xp = Object.values(logs).reduce(
   104→    (a, log) => a + dayCount(log) * 10 + (dayCount(log) >= 5 ? 25 : 0),
   105→    0
   106→  );
   107→  const unlocked: string[] = [];
   108→  const add = (id: string, cond: boolean) => cond && unlocked.push(id);
   109→  add("first-prayer", total >= 1);
   110→  add("full-day", fullDays >= 1);
   111→  add("streak-3", bestVal >= 3);
   112→  add("streak-7", bestVal >= 7);
   113→  add("streak-30", bestVal >= 30);
   114→  add("prayers-50", total >= 50);
   115→  add("prayers-100", total >= 100);
   116→  add("fajr-master", fajrCount >= 5);
   117→  add("week-perfect", fullDays >= 7);
   118→  add("level-5", xp >= 400);
   119→  // أوسمة التاسكات والتايمر (الدمج مع Challenge Planner)
   120→  if (planner) {
   121→    const m = plannerMetrics(planner);
   122→    add("first-task", m.done >= 1);
   123→    add("task-day", m.fullDays >= 1);
   124→    add("study-1h", m.study >= 3600);
   125→    add("tasks-streak-3", m.best >= 3);
   126→    add("units-10", m.units >= 10);
   127→    add("study-10h", m.study >= 36000);
   128→    add("task-half", m.pct >= 50);
   129→    add("units-50", m.units >= 50);
   130→    add("study-25h", m.study >= 90000);
   131→    add("course-done", m.courseDone);
   132→    add("all-tasks", m.total > 0 && m.done === m.total);
   133→  }
   134→  return unlocked;
   135→}
